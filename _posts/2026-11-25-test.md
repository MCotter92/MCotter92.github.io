require "json"

module Palette
  ACCENTS = %i[blue red green].freeze

  class Swatch
    attr_reader :name, :hex

    def initialize(name, hex = "#000000")
      @name = name
      @hex  = hex
    end

    def rgb
      hex.scan(/[0-9a-f]{2}/i).map { |pair| pair.to_i(16) }
    end

    def to_s = "#{name}: #{hex}\n"
  end
end

## Python

Decorators, builtins, types, and numbers in several bases:

```python
from dataclasses import dataclass


@dataclass(frozen=True)
class Color:
    name: str
    value: int = 0xFF00AA

    def channels(self) -> tuple[int, int, int]:
        """Split the packed value into r, g, b."""
        return (self.value >> 16, (self.value >> 8) & 0xFF, self.value & 0xFF)


colors = [Color("red", 0xD66A62), Color("green", 0x8FAB72)]
print(len(colors), isinstance(colors[0], Color), 3.14, 0b1010, None)
```

## JavaScript

```javascript
const ACCENT = "#7faabe";

export async function applyTheme(root = document.documentElement) {
  const vars = { accent: ACCENT, ratio: 1.5, enabled: true };
  for (const [key, value] of Object.entries(vars)) {
    root.style.setProperty(`--${key}`, String(value));
  }
  return /^#[0-9a-f]{6}$/i.test(ACCENT) ? null : undefined;
}
```
